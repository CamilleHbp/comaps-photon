cmake_minimum_required(VERSION 3.22.1)
project(valhalla_api_tests LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 23)
get_filename_component(COMAPS_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(JANSSON_BUILD_DOCS OFF CACHE BOOL "" FORCE)
set(JANSSON_BUILD_MAN OFF CACHE BOOL "" FORCE)
set(JANSSON_EXAMPLES OFF CACHE BOOL "" FORCE)
set(JANSSON_INSTALL OFF CACHE BOOL "" FORCE)
set(JANSSON_WITHOUT_TESTS ON CACHE BOOL "" FORCE)
add_subdirectory("${COMAPS_ROOT}/3party/jansson/jansson" jansson)
add_executable(valhalla_api_tests valhalla_api_tests.cpp "${COMAPS_ROOT}/libs/routing/valhalla_api.cpp")
target_include_directories(valhalla_api_tests PRIVATE "${COMAPS_ROOT}/libs" "${CMAKE_CURRENT_BINARY_DIR}/jansson/include")
target_link_libraries(valhalla_api_tests PRIVATE jansson)
enable_testing()
add_test(NAME valhalla_api_tests COMMAND valhalla_api_tests "${COMAPS_ROOT}/data/test_data/valhalla/route.json")
